Вот несколько вариантов функции fan-out, которая принимает входной канал и возвращает массив выходных каналов:

## Вариант 1: Простой fan-out (равномерное распределение)

```go
// FanOutSimple создает N выходных каналов и распределяет сообщения
// из входного канала по ним в round-robin порядке
func FanOutSimple(input <-chan interface{}, numOutputs int) []<-chan interface{} {
    outputs := make([]chan interface{}, numOutputs)
    results := make([]<-chan interface{}, numOutputs)
    
    // Создаем буферизованные каналы
    for i := 0; i < numOutputs; i++ {
        outputs[i] = make(chan interface{}, 100)
        results[i] = outputs[i]
    }
    
    go func() {
        defer func() {
            // Закрываем все выходные каналы после завершения входного
            for i := 0; i < numOutputs; i++ {
                close(outputs[i])
            }
        }()
        
        i := 0
        for value := range input {
            // Round-robin распределение
            outputs[i] <- value
            i = (i + 1) % numOutputs
        }
    }()
    
    return results
}
```

## Вариант 2: Fan-out с возможностью выбора стратегии распределения

```go
// DistributionStrategy определяет стратегию распределения сообщений
type DistributionStrategy int

const (
    RoundRobin DistributionStrategy = iota
    Broadcast
    HashBased
)

// FanOut распределяет сообщения из входного канала в выходные каналы
// согласно выбранной стратегии
func FanOut(
    input <-chan interface{},
    numOutputs int,
    strategy DistributionStrategy,
) []<-chan interface{} {
    outputs := make([]chan interface{}, numOutputs)
    results := make([]<-chan interface{}, numOutputs)
    
    // Инициализация каналов
    for i := 0; i < numOutputs; i++ {
        outputs[i] = make(chan interface{}, 100)
        results[i] = outputs[i]
    }
    
    // Запуск распределителя в зависимости от стратегии
    switch strategy {
    case RoundRobin:
        go fanOutRoundRobin(input, outputs)
    case Broadcast:
        go fanOutBroadcast(input, outputs)
    case HashBased:
        go fanOutHashBased(input, outputs)
    }
    
    return results
}

func fanOutRoundRobin(input <-chan interface{}, outputs []chan interface{}) {
    defer closeAll(outputs)
    
    i := 0
    for value := range input {
        outputs[i] <- value
        i = (i + 1) % len(outputs)
    }
}

func fanOutBroadcast(input <-chan interface{}, outputs []chan interface{}) {
    defer closeAll(outputs)
    
    for value := range input {
        // Отправляем во ВСЕ выходные каналы
        for _, out := range outputs {
            out <- value
        }
    }
}

func fanOutHashBased(input <-chan interface{}, outputs []chan interface{}) {
    defer closeAll(outputs)
    
    for value := range input {
        // Используем хеш для детерминированного распределения
        hash := fmt.Sprintf("%v", value)
        idx := int(hash[0]) % len(outputs)
        outputs[idx] <- value
    }
}

func closeAll(channels []chan interface{}) {
    for _, ch := range channels {
        close(ch)
    }
}
```

## Вариант 4: Компактная универсальная реализация

```go
// SplitChannel разделяет один канал на несколько
func SplitChannel[T any](source <-chan T, n int) []<-chan T {
    // Создаем выходные каналы
    outputs := make([]chan T, n)
    for i := range outputs {
        outputs[i] = make(chan T)
    }
    
    // Запускаем распределитель
    go func() {
        defer func() {
            // Закрываем все выходные каналы
            for _, out := range outputs {
                close(out)
            }
        }()
        
        // Распределяем сообщения round-robin
        i := 0
        for msg := range source {
            outputs[i] <- msg
            i = (i + 1) % n
        }
    }()
    
    // Конвертируем в read-only каналы для возврата
    result := make([]<-chan T, n)
    for i, ch := range outputs {
        result[i] = ch
    }
    
    return result
}
```

## Пример использования:

```go
func main() {
    // Создаем входной канал
    input := make(chan int, 10)
    
    // Наполняем данными
    go func() {
        for i := 1; i <= 100; i++ {
            input <- i
        }
        close(input)
    }()
    
    // Разделяем на 3 канала
    outputs := SplitChannel(input, 3)
    
    // Читаем из выходных каналов
    var wg sync.WaitGroup
    for i, out := range outputs {
        wg.Add(1)
        go func(id int, ch <-chan int) {
            defer wg.Done()
            for val := range ch {
                fmt.Printf("Worker %d: %d\n", id, val)
            }
        }(i, out)
    }
    
    wg.Wait()
}
```

**Ключевые моменты:**
1. Функция всегда возвращает каналы только для чтения (`<-chan`)
2. Все выходные каналы закрываются после закрытия входного канала
3. По умолчанию используется round-robin стратегия распределения
4. Буферизация каналов предотвращает блокировки
