Вот несколько вариантов реализации паттерна **fan-out** в Go:

## 1. Базовый Fan-out (разделение работы между воркерами)

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

func main() {
	// Входной канал с данными для обработки
	jobs := make(chan int, 100)
	results := make(chan string, 100)

	// Генератор данных
	go func() {
		for i := 1; i <= 20; i++ {
			jobs <- i
		}
		close(jobs)
	}()

	// Количество воркеров
	numWorkers := 5
	var wg sync.WaitGroup

	// Запуск воркеров (fan-out)
	for i := 1; i <= numWorkers; i++ {
		wg.Add(1)
		go worker(i, jobs, results, &wg)
	}

	// Закрываем results после завершения всех воркеров
	go func() {
		wg.Wait()
		close(results)
	}()

	// Сбор результатов (fan-in)
	for result := range results {
		fmt.Println(result)
	}
}

func worker(id int, jobs <-chan int, results chan<- string, wg *sync.WaitGroup) {
	defer wg.Done()
	
	for job := range jobs {
		// Имитация обработки
		time.Sleep(100 * time.Millisecond)
		results <- fmt.Sprintf("Worker %d processed job %d", id, job)
	}
}
```

## 2. Fan-out с ограничением скорости (rate limiting)

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

func main() {
	source := make(chan int)
	const numWorkers = 3

	// Генерация данных
	go func() {
		for i := 1; i <= 10; i++ {
			source <- i
		}
		close(source)
	}()

	// Каналы для сбора результатов
	var results []chan string
	for i := 0; i < numWorkers; i++ {
		results = append(results, make(chan string, 10))
	}

	var wg sync.WaitGroup

	// Запуск воркеров с fan-out
	for i := 0; i < numWorkers; i++ {
		wg.Add(1)
		go func(workerID int, outChan chan<- string) {
			defer wg.Done()
			defer close(outChan)
			
			// Ограничение скорости: не более 1 операции в 200ms
			limiter := time.Tick(200 * time.Millisecond)
			
			for n := range source {
				<-limiter // Ожидаем тик для rate limiting
				outChan <- fmt.Sprintf("Worker %d: processed %d", workerID, n)
			}
		}(i, results[i])
	}

	// Fan-in результатов
	merged := fanIn(results...)
	
	// Вывод результатов
	for msg := range merged {
		fmt.Println(msg)
	}
	
	wg.Wait()
}

// Fan-in функция для объединения нескольких каналов
func fanIn(channels ...chan string) chan string {
	var wg sync.WaitGroup
	multiplexed := make(chan string)

	// Чтение из каждого канала
	for _, ch := range channels {
		wg.Add(1)
		go func(c <-chan string) {
			defer wg.Done()
			for msg := range c {
				multiplexed <- msg
			}
		}(ch)
	}

	// Закрываем multiplexed после завершения всех чтений
	go func() {
		wg.Wait()
		close(multiplexed)
	}()

	return multiplexed
}
```

## 3. Fan-out для обработки разного типа задач

```go
package main

import (
	"fmt"
	"math/rand"
	"sync"
	"time"
)

type Task struct {
	ID     int
	Type   string
	Data   interface{}
	Result chan string
}

func main() {
	const numWorkers = 4
	
	// Создаем специализированных воркеров
	workers := map[string]chan Task{
		"fast":   make(chan Task, 10),
		"slow":   make(chan Task, 10),
		"io":     make(chan Task, 10),
		"cpu":    make(chan Task, 10),
	}

	var wg sync.WaitGroup

	// Запуск специализированных воркеров
	for taskType, taskChan := range workers {
		wg.Add(1)
		go func(tType string, ch <-chan Task) {
			defer wg.Done()
			processWorker(tType, ch)
		}(taskType, taskChan)
	}

	// Диспетчер задач (распределяет задачи по типам)
	tasks := generateTasks(20)
	go dispatchTasks(tasks, workers)

	// Сбор результатов
	var results []string
	for _, task := range tasks {
		result := <-task.Result
		results = append(results, result)
		fmt.Println(result)
	}

	// Закрываем каналы воркеров
	for _, ch := range workers {
		close(ch)
	}

	wg.Wait()
	fmt.Printf("\nОбработано задач: %d\n", len(results))
}

func generateTasks(count int) []Task {
	var tasks []Task
	for i := 1; i <= count; i++ {
		taskTypes := []string{"fast", "slow", "io", "cpu"}
		taskType := taskTypes[rand.Intn(len(taskTypes))]
		
		tasks = append(tasks, Task{
			ID:     i,
			Type:   taskType,
			Data:   fmt.Sprintf("data-%d", i),
			Result: make(chan string, 1),
		})
	}
	return tasks
}

func dispatchTasks(tasks []Task, workers map[string]chan Task) {
	for _, task := range tasks {
		// Fan-out: распределяем задачи по соответствующим воркерам
		if ch, exists := workers[task.Type]; exists {
			ch <- task
		} else {
			// Если тип не найден, отправляем в дефолтный
			workers["fast"] <- task
		}
	}
}

func processWorker(workerType string, tasks <-chan Task) {
	for task := range tasks {
		// Имитация разной скорости обработки
		var processTime time.Duration
		switch workerType {
		case "fast":
			processTime = 50 * time.Millisecond
		case "slow":
			processTime = 300 * time.Millisecond
		case "io":
			processTime = 200 * time.Millisecond
		case "cpu":
			processTime = 150 * time.Millisecond
		}

		time.Sleep(processTime)
		task.Result <- fmt.Sprintf("%s worker processed task %d in %v", 
			workerType, task.ID, processTime)
		close(task.Result)
	}
}
```

## 4. Fan-out с пулом воркеров и балансировкой нагрузки

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

type WorkerPool struct {
	workers []chan int
	results chan string
	wg      sync.WaitGroup
}

func NewWorkerPool(numWorkers int) *WorkerPool {
	wp := &WorkerPool{
		workers: make([]chan int, numWorkers),
		results: make(chan string, 100),
	}

	// Создаем и запускаем воркеров
	for i := 0; i < numWorkers; i++ {
		wp.workers[i] = make(chan int, 10)
		wp.wg.Add(1)
		go wp.startWorker(i, wp.workers[i])
	}

	return wp
}

func (wp *WorkerPool) startWorker(id int, jobs <-chan int) {
	defer wp.wg.Done()
	
	for job := range jobs {
		time.Sleep(100 * time.Millisecond) // Имитация работы
		wp.results <- fmt.Sprintf("Worker %d processed %d", id, job)
	}
}

func (wp *WorkerPool) Dispatch(job int) {
	// Простой round-robin балансировщик
	workerID := job % len(wp.workers)
	wp.workers[workerID] <- job
}

func (wp *WorkerPool) Close() {
	// Закрываем все каналы воркеров
	for _, ch := range wp.workers {
		close(ch)
	}
	wp.wg.Wait()
	close(wp.results)
}

func (wp *WorkerPool) Results() <-chan string {
	return wp.results
}

func main() {
	// Создаем пул из 4 воркеров
	pool := NewWorkerPool(4)

	// Горутина для отправки заданий
	go func() {
		for i := 1; i <= 20; i++ {
			pool.Dispatch(i)
		}
		pool.Close()
	}()

	// Чтение результатов
	for result := range pool.Results() {
		fmt.Println(result)
	}

	fmt.Println("All tasks completed")
}
```

## Ключевые концепции Fan-out в Go:

1. **Разделение работы**: Один входной канал разделяется между несколькими горутинами
2. **Балансировка нагрузки**: Задачи распределяются между воркерами
3. **Параллелизм**: Несколько задач выполняются одновременно
4. **Специализация**: Можно создавать специализированных воркеров для разных типов задач

**Советы по использованию:**
- Используйте буферизованные каналы для предотвращения блокировок
- Всегда закрывайте каналы, когда они больше не нужны
- Используйте `sync.WaitGroup` для ожидания завершения всех воркеров
- Рассмотрите контексты (`context.Context`) для отмены операций
