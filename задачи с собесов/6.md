```text
Напишите функцию mergeSorted, которая принимает на вход два "отсортированных" канала и возвращает результирующий
отсортированный канал.
```

```go
package main

import "fmt"

/*
Беру первые значения из двух каналов.
Иду в цикле по двум каналам одновременно до тех пор пока у одного из них не закончатся элементы
Заполняю результирующий канал при помощи сравнения значений
Если значение запушил в канал то его сразу же обновляю, читая следующее значение из канала

Как только элементы в одном из каналов закончатся, прохожу остаток в одном из двух каналов
и заполняю последние элементы
*/
func mergeSorted(a, b <-chan int) <-chan int {
	result := make(chan int)

	go func() {
		defer close(result)

		v1, ok1 := <-a
		v2, ok2 := <-b

		for ok1 && ok2 {
			if v1 < v2 {
				result <- v1
				v1, ok1 = <-a
			} else {
				result <- v2
				v2, ok2 = <-b
			}
		}
		if ok1 {
			result <- v1
			v1, _ = <-a
		}

		if ok2 {
			result <- v2
			v2, _ = <-b
		}

	}()

	return result
}

func fillChanA(c chan int) {
	c <- 1
	c <- 2
	c <- 4
	close(c)
}
func fillChanB(c chan int) {
	c <- -1
	c <- 4
	c <- 5
	close(c)
}
func main() {
	a, b := make(chan int), make(chan int)
	go fillChanA(a)
	go fillChanB(b)
	c := mergeSorted(a, b)
	for val := range c {
		fmt.Printf("%d ", val)
	}
}
```
