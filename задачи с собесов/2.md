/*

1.Делаем структуру в которой будем хранить и ошибку и сообщение.
При асинхронной работе c fakeDownload записываем весь результат. И только потом уже работаем с ним - сортируем
результаты на удавшиеся и нет.
Ошибки обрабатываем так: прохожусь циклом по каналу, если ошибка не nil то делаю Join к одной

func main() {
	msgs, err, m := download([]string{
		"https://example.com/e25e26d3-6aa3-4d79-9ab4-fc9b71103a8c.xml",
		"https://example.com/a601590e-31c1-424a-8ccc-decf5b35c0f6.xml",
		"https://example.com/1cf0dd69-a3e5-4682-84e3-dfe22ca771f4.xml",

		"https://example.com/b6ed16d7-cb3d-4cba-b81a-01a789d3a914.xml",
	})
	if err != nil {
		log.Println(m)
	}
	fmt.Println(msgs)
}

type Result struct {
	msg string
	err error
}

func download(urls []string) ([]string, error, int) {
	var resultError error
	messageResult := make([]string, len(urls))
	ch := make(chan Result)

	wg := sync.WaitGroup{}
	wg.Add(len(urls))
	for _, url := range urls {
		go func() {
			defer wg.Done()
			ch <- fakeDownload(url)
		}()
	}

	go func() {
		wg.Wait()
		close(ch)
	}()
	n := 0
	for result := range ch {

		if result.err != nil {

			resultError = errors.Join(resultError, result.err)
			n++
		}

		messageResult = append(messageResult, result.msg)
	}
	return messageResult, resultError, n
}

func fakeDownload(url string) Result {

	randomNumber := rand.Float64()
	if randomNumber < 0.5 {
		result := Result{msg: "", err: errors.New("500")}
		return result
	}

	result := Result{msg: "downloaded", err: nil}

	return result
}
