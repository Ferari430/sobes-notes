```text
Напишите функции merge и fillChan .
Функция fillChan:
на вход получает целое число n;
возвращает канал;
пишет в этот канал n чисел от 0 до n-1.
Функция merge:
получает на вход массив каналов cs;
возвращает канал;
параллельно читает из каждого канала из cs и пишет полученное значение в возвращаемый канал.
```

```go
package main

import (
	"fmt"
	"sync"
)

func main() {
	a := fillChan(2)
	b := fillChan(3)
	c := fillChan(4)
	d := merge(a, b, c)
	for v := range d {
		fmt.Println(v)
	}
}

func fillChan(n int) <-chan int {
	result := make(chan int)

	go func() {
		for i := 0; i < n; i++ {
			result <- i
		}
		close(result)
	}()

	return result
}

/*
необходимо быть очень внимательным,если в функцию передается массив каналов. Потому что запись
for ch := range chans{} подразумевает итерирование по индексу массива а не по каналам!
*/
func merge(chans ...<-chan int) <-chan int {

	wg := sync.WaitGroup{}
	result := make(chan int)
	wg.Add(len(chans))
	for _, ch := range chans {
		go func() {
			defer wg.Done()
			for v := range ch {
				result <- v
			}
		}()
	}
	go func() {
		wg.Wait()
		close(result)
	}()
	return result
}
```
