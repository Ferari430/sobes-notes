```go
package main

import (
	"fmt"
	"time"
)

// Need to show solution

func producer(ch chan<- int) {
	for {
		ch <- 1
		time.Sleep(time.Second)
	}
}

/*
Пусть первый канал будет с приоритетом. Поэтому во втором кейсе снова проверяем пришло
ли что-то в первый канал. Если реально ничего не пришло в первый, то выходим по дефолту и
продолжаем логику со вторым каналом.
*/
func main() {
	ch1 := make(chan int) // more prioritized
	ch2 := make(chan int)

	go producer(ch1)
	go producer(ch2)

	for {
		select {
		case value := <-ch1:
			fmt.Println(value)
			return
		case value := <-ch2:
			select {
			case v := <-ch1:
				fmt.Println(v)
				return
			default:
			}
			fmt.Println(value)
		}
	}
}
```

```go
package main

import "fmt"

/*
Ещё один способ приоритезации - добавить несколько одинаковых кейсов.
В этом примере fmt.Println(ch1Value / ch2Value) выведет 1,98, то есть первый канал в два раза
чаще используется
*/
func main() {
	ch1 := make(chan struct{}, 1)
	ch2 := make(chan struct{}, 1)

	close(ch1)
	close(ch2)

	ch1Value := 0.0
	ch2Value := 0.0

	for i := 0; i < 100000; i++ {
		select {
		case <-ch1:
			ch1Value++
		case <-ch1:
			ch1Value++
		case <-ch2:
			ch2Value++
		}
	}

	fmt.Println(ch1Value / ch2Value)
}
```
