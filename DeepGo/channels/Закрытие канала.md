package main

import (
	"fmt"
	"log"
	"time"
)

/*
Что будет если не закрыть канал?
Мы пишем в канал строку и не закрываем канал. Горутина прочитает все записанные в канал значения
Но не сможет завершиться, потому что range ждет закрытие канала. Поэтому имеем утечку горутины
То есть она работает до тех пор пока работает main, канал так же не очистится потому что на канал
ссылается горутина.
*/
func main() {
	doWork := func(strings <-chan string) {
		go func() {
			for str := range strings {
				fmt.Println(str)
			}

			log.Println("doWork exited")
		}()
	}

	strings := make(chan string)
	doWork(strings)
	strings <- "Test"
	//close(strings)
	time.Sleep(time.Second)
	fmt.Println("Done")
}

package main

// First-response-wins strategy

/*
Запускаем 5 горутин и ждем пока хотя бы одна запишет в канал значение. После первой записи
сразу же выходим из функции.
Остальные 4 горутины заблокируются, потому что из канала уже никто не читает.
*/
func request() int {
	ch := make(chan int)
	for i := 0; i < 5; i++ {
		go func() {
			ch <- i // 4 goroutines will be blocked
		}()
	}
	//ждем пока хотя бы одна горутина запишет в канал и сразу читаем значение из
	// канала которое уходит наружу
	return <-ch
}

/*
Пишем в канал, потом читаем из него и отдаем значение на ружу, все как и было.
Но остальные 4 горутины запишут в канал свои значения и завершат свою работу, на канал
больше никто не будет ссылаться, поэтому он с данными в буфере сотрется сборщиком мусора.
*/
func request1() int {
	ch := make(chan int, 4)
	for i := 0; i < 5; i++ {
		go func() {
			ch <- i
		}()
	}
	return <-ch
}
