```go
package main

import "fmt"

/*
	На что будет относиться continue и break?
	continue and break относятся к select.
	Чтобы отнести их к for{} необходимо писать
	loop:
		for{
			break loop
		}

*/

func main() {
	data := make(chan int)
	go func() {
		for i := 1; i <= 4; i++ {
			data <- i
		}
		close(data)
	}()

	for {
		value := 0
		opened := true

		select {
		case value, opened = <-data:
			if value == 2 {
				continue
			} else if value == 3 {
				break
			}

			if !opened {
				return
			}
		}

		fmt.Println(value)
	}
}
```

```go
package main

import (
	"fmt"
	"time"
)

/*
Тут все довольно просто. В селекте ждем результат из одной из двух горутин.
Закрывать канал не нужно. Если в писателе закроем нанал то из него нельзя ничего прочитать
в мэйне.
*/
func async1() chan string {
	ch := make(chan string)
	go func() {
		time.Sleep(1 * time.Second)
		ch <- "async1 result"
	}()
	return ch
}

func async2() chan string {
	ch := make(chan string)
	go func() {
		time.Sleep(1 * time.Second)
		ch <- "async2 result"
	}()
	return ch
}

func main() {
	ch1 := async1()
	ch2 := async2()

	select {
	case result := <-ch1:
		fmt.Println(result)
	case result := <-ch2:
		fmt.Println(result)
	}
}
```

```go
package main

import (
	"fmt"
	"sync"
)

func producer(ch chan int) {
	defer close(ch)
	for i := 0; i < 5; i++ {
		ch <- i
	}
}

/*
	Концепция продюсер - консьюмер: один пишет другой читает.
range по каналу идет до тех пор пока он не будет закрыт.
range эквивалентен for{select{}}
*/

func consumer(ch chan int) {

	/*
		for {
			select {
			case value, opened := <-ch:
				if !opened {
					return
				}

				fmt.Println(value)
			}
		}
	*/

	for value := range ch { // syntax sugar
		fmt.Println(value)
	}
}

func main() {
	ch := make(chan int)
	wg := sync.WaitGroup{}
	wg.Add(2)

	go func() {
		defer wg.Done()
		producer(ch)
	}()

	go func() {
		defer wg.Done()
		consumer(ch)
	}()

	wg.Wait()
}
```

```go
package main

import (
	"fmt"
	"time"
)

/*
Создается две функции которые после сна пишут в свои каналы 10 и 20.
Сколько времени будет выполняться код? Ответ: 4с,а не 2с.
Потому что при вызове Process(<-FetchData1(), <-FetchData2()) аргументы вычисляются последовательно.
То есть вторая функция ждет завершения другой
*/
func FetchData1() chan int {
	ch := make(chan int)
	go func() {
		time.Sleep(time.Second * 2)
		ch <- 10
	}()

	return ch
}

func FetchData2() chan int {
	ch := make(chan int)
	go func() {
		time.Sleep(time.Second * 2)
		ch <- 20
	}()

	return ch
}

func Process(value1, value2 int) {
	// Processing...
}

func main() {
	start := time.Now()
	Process(<-FetchData1(), <-FetchData2())
	fmt.Println(time.Now().Sub(start))
}
//Исправленный вариант
func main1() {
	start := time.Now()
	ch1 := FetchData1()
	ch2 := FetchData1()
	Process(<-ch1, <-ch2)
	fmt.Println(time.Now().Sub(start))
}
```
