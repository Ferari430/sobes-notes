```go
package main

/*
Чтение из нил мапы - не ломает код, пустой вывод
*/
func readFromNilMap() {
	var data map[int]int
	_ = data[100]
}

/*
Удаление из нил мапы - не ломает код, пустой вывод
*/
func deleteFromNilMap() {
	var data map[int]int
	delete(data, 100)
}

/*
Запись в нил мапу - паника panic: assignment to entry in nil map
*/
func writeToNilMap() {
	var data map[int]int
	data[100] = 100
}

/*
range по нил мапе - так же не ломает код, как и чтение из нил мапы
*/
func rangeByNilMap() {
	var data map[int]int
	for range data {
	}
}

/*
Можно переписать значение по одному и тому же ключу.
*/
func rewriteExistingKey() {
	data := make(map[int]int)
	data[100] = 500
	data[100] = 1000
}

/*
Удаление значения которого нет в мапе - работает валидно, не падает
*/
func deleteNonExistingKey() {
	data := make(map[int]int)
	delete(data, 100)
}

func main() {
	writeToNilMap()
}
```

```go
package main

import "fmt"

/*
	data[value1+value2] = struct{}{} Добавляем в мапу значение по ключу 0,6+0,3

потом создаем отдельную переменную var result float32 = 0.9
и пытаемся по этому значению найти объект. Найдет?
Ответ: нет, потому что при работе c float мы теряем точность. Даже проверка

	if value1+value2 == result выдаст false.
*/
func main() {
	var value1 float32 = 0.3
	var value2 float32 = 0.6
	data := make(map[float32]struct{})
	data[value1+value2] = struct{}{}

	var result float32 = 0.9
	_, found := data[result]
	fmt.Println(found)
}
```

```go
package main

import "fmt"

/*
Вопрос будет ли порядок при итерации по мапе в цикле?
Ответ: нет, при каждом range в функции вывод будет разный, даже при 2range.
Но при Println порядок будет соблюдаться. Всегда.
Потому что Println под капотом сортирует по ключам.
*/

func main() {
	data := map[int]struct{}{
		1: {}, 2: {}, 3: {}, 4: {}, 5: {},
	}

	for key := range data {
		fmt.Print(key, " ")
	}

	fmt.Println()
	fmt.Println(data)
}
```

```go
package main

import "fmt"

/*
Почему в мапе нигде не будет присвоено 1000?
Потому что я достаю копию элемента из мапы и ей присваиваю 1000.
Но в самой мапе ничего не меняется
*/
func main() {
	data := map[int]int{1: 1, 2: 2, 3: 3}
	for _, value := range data {
		value = 1000//copy
		_ = value
	}

	fmt.Println(data)
```

```go
package main

import "fmt"

/*
итерируемся по мапе и на каждой итерации добавляем в мапу значение.
Сколько элементов по итогу цикла будет в мапе?
Ответ: не определенное значение.
В момент итерации по мапе мы можем вставить элемент как слева, так и справа от указателя на ключ
на котором мы сейчас находимся.
Если вставим новый ключ слева от указателя то длина мапы не поменяется.
Если вставим справа от указателя то длина мапы поменяется и range пройдет ещё
одну дополнительную итерацию, что приведет в конечном итоге к
неопределенному поведению мапы.

*/

func main() {
	data := map[int]struct{}{
		0: {},
		1: {},
		2: {},
	}

	for key := range data {
		fmt.Println(key)
		data[10+key] = struct{}{}
	} //дополняем мапу

	for key := range data {
		fmt.Print(key)
		fmt.Print(" ")
	}

	fmt.Println()
	//fmt.Println(data)
}
```

```go
package main

import "fmt"

func main() {
	// remove all elements
	first := map[int]int{1: 1, 2: 2, 3: 3}
	first = nil //удаление всего словаря
	fmt.Println(first, " : ", len(first))

	// keep allocated memory
	second := map[int]int{1: 1, 2: 2, 3: 3}
	for key := range second {
		delete(second, key) //удаляем элементы но бакеты остаются жить
	}
	fmt.Println(second, " : ", len(second))

	// keep allocated memory
	third := map[int]int{1: 1, 2: 2, 3: 3}
	clear(third) //удаляем элементы но бакеты остаются жить
	fmt.Println(third, " : ", len(third))
}
```

```go
package main

import (
	"fmt"
	"runtime"
)

func printAllocs() {
	var m runtime.MemStats
	runtime.ReadMemStats(&m)
	fmt.Printf("%d MB\n", m.Alloc/1024/1024)
}

/*
	тут заполняем мапу на 300МБ. Потом чистим объекты в ней но мапа

все так же остается большой даже после delete:было 314МБ стало 287МБ
*/
func main() {
	printAllocs()
	data := make(map[int][128]byte) // try to use 129
	printAllocs()

	for i := 0; i < 1_000_000; i++ {
		data[i] = [128]byte{}
	}

	printAllocs()

	for i := 0; i < 1_000_000; i++ {
		delete(data, i)
	}

	runtime.GC()
	printAllocs()
	runtime.KeepAlive(data)
}
```

```go
package main

type client struct {
	name    string
	surname string
}

/*
пытаемся достать значение по ключу из мапы и обновить то что лежит внутри

	// data[id].surname = surname -> compilation error

Но можно в мапе хранить не значения структуры а указатели на структуру.
Тогда изменение валидно
*/
func updateSurname(data map[int]client, id int, surname string) {
	object := data[id] // copy
	object.surname = surname
	data[id] = object // copy

	// data[id].surname = surname -> compilation error
}

func updateSurnameByPointer(data map[int]*client, id int, surname string) {
	data[id].surname = surname
}

func updateSurnamesSlice(data map[int][]string, id int, surname string) {
	data[id][5] = surname
}
```
